<?php
declare(strict_types=1);
mb_internal_encoding('UTF-8');
mb_language('ja');

$CFG = require __DIR__ . '/config.php';
if (is_file(__DIR__ . '/config.local.php')) $CFG = array_replace_recursive($CFG, require __DIR__ . '/config.local.php');

function cfg(string $k) { global $CFG; return $CFG[$k] ?? null; }
function h($s): string { return htmlspecialchars((string)$s, ENT_QUOTES, 'UTF-8'); }
function yen(int $n): string { return '¥' . number_format($n); }
function tax_in(int $gross): int { $r = (int)cfg('TAX_RATE'); return (int)round($gross * $r / (100 + $r)); }
function jdate(string $ymd): string { return date('Y年n月j日', strtotime($ymd)); }

function data_dir(): string {
  $d = cfg('DATA_DIR');
  if (!is_dir($d)) @mkdir($d, 0700, true);
  if (!is_dir($d) || !is_writable($d)) {
    $d = __DIR__ . '/data';
    if (!is_dir($d)) @mkdir($d, 0700, true);
    if (!file_exists($d . '/.htaccess')) @file_put_contents($d . '/.htaccess', "Require all denied\nDeny from all\n");
  }
  return $d;
}

/** 注文一覧をロック付きで読み書きする。$fn(array &$orders) が true を返せば保存 */
function with_orders(callable $fn) {
  $file = data_dir() . '/orders.json';
  $fp = fopen($file, 'c+');
  if (!$fp) throw new RuntimeException('data not writable');
  flock($fp, LOCK_EX);
  $raw = stream_get_contents($fp);
  $orders = $raw ? (json_decode($raw, true) ?: []) : [];
  $ret = $fn($orders);
  if ($ret === true) {
    ftruncate($fp, 0); rewind($fp);
    fwrite($fp, json_encode($orders, JSON_UNESCAPED_UNICODE | JSON_PRETTY_PRINT));
    fflush($fp);
  }
  flock($fp, LOCK_UN); fclose($fp);
  return $orders;
}
function all_orders(): array { $o = with_orders(fn(&$x) => false); usort($o, fn($a,$b)=>strcmp($b['created_at'],$a['created_at'])); return $o; }
function find_order(string $id): ?array { foreach (with_orders(fn(&$x)=>false) as $o) if ($o['id'] === $id) return $o; return null; }
function find_by_token(string $t): ?array { if ($t === '') return null; foreach (with_orders(fn(&$x)=>false) as $o) if (hash_equals($o['token'], $t)) return $o; return null; }
function update_order(string $id, callable $fn): ?array {
  $out = null;
  with_orders(function (&$orders) use ($id, $fn, &$out) {
    foreach ($orders as &$o) if ($o['id'] === $id) { $fn($o); $out = $o; return true; }
    return false;
  });
  return $out;
}

/** 連番: INV-2026-0001 / DN-2026-0001 */
function next_number(string $prefix): string {
  $file = data_dir() . '/seq.json';
  $fp = fopen($file, 'c+'); flock($fp, LOCK_EX);
  $raw = stream_get_contents($fp); $seq = $raw ? (json_decode($raw, true) ?: []) : [];
  $y = date('Y'); $key = $prefix . $y;
  $seq[$key] = ($seq[$key] ?? 0) + 1;
  ftruncate($fp, 0); rewind($fp); fwrite($fp, json_encode($seq)); fflush($fp); flock($fp, LOCK_UN); fclose($fp);
  return sprintf('%s-%s-%04d', $prefix, $y, $seq[$key]);
}

function send_mail(string $to, string $subject, string $body, ?string $bcc = null, ?string $cc = null): bool {
  $from = cfg('MAIL_FROM'); $name = mb_encode_mimeheader(cfg('MAIL_FROM_NAME'), 'UTF-8');
  $h = "From: {$name} <{$from}>\r\nReply-To: {$from}\r\nContent-Type: text/plain; charset=UTF-8\r\nContent-Transfer-Encoding: 8bit";
  if ($cc) $h .= "\r\nCc: {$cc}";
  if ($bcc) $h .= "\r\nBcc: {$bcc}";
  return @mail($to, mb_encode_mimeheader($subject, 'UTF-8'), $body, $h, '-f' . $from);
}

function bank_text(): string {
  $b = cfg('BANK');
  if (empty($b['bank']) || empty($b['number'])) return '（振込先は別途ご案内します）';
  return "{$b['bank']} {$b['branch']}\n{$b['type']} {$b['number']}\n{$b['holder']}";
}

function course_of(array $o): array { return cfg('COURSES')[$o['course']]; }

/** 注文作成 */
function create_order(array $in): array {
  $c = cfg('COURSES')[$in['course']];
  $now = date('Y-m-d H:i:s');
  $o = [
    'id'         => 'W' . date('ymd') . '-' . strtoupper(substr(bin2hex(random_bytes(3)), 0, 5)),
    'token'      => bin2hex(random_bytes(16)),
    'created_at' => $now,
    'status'     => 'invoiced',          // invoiced → paid → shipped (/ cancelled)
    'course'     => $in['course'],
    'amount'     => $c['price'] + ($in['photo'] ? (int)$c['photo'] : 0),
    'photo'      => $in['photo'] ? 1 : 0,
    'photo_fee'  => $in['photo'] ? (int)$c['photo'] : 0,
    'course_price' => $c['price'],
    'name'       => $in['name'],
    'company'    => $in['company'],
    'email'      => $in['email'],
    'tel'        => $in['tel'],
    'zip'        => $in['zip'],
    'address'    => $in['address'],
    'note'       => $in['note'],
    'invoice_no' => next_number('INV'),
    'invoice_date' => date('Y-m-d'),
    'due_date'   => date('Y-m-d', strtotime('+' . (int)cfg('DUE_DAYS') . ' days')),
    'paid_at'    => null, 'ship_deadline' => null,
    'delivery_no' => null, 'delivery_date' => null, 'delivery_items' => null, 'delivery_total' => null,
    'shipped_at' => null, 'tracking' => '',
    'log'        => [[$now, '受注・請求書発行']],
  ];
  with_orders(function (&$orders) use ($o) { $orders[] = $o; return true; });
  return $o;
}

function doc_url(array $o, string $kind): string { return cfg('BASE_URL') . '/doc.php?k=' . $kind . '&t=' . $o['token']; }

/** 請求書メールの件名・本文テンプレート（自動送信とメールフォームの両方で使う） */
function invoice_mail_template(array $o): array {
  $c = course_of($o); $co = cfg('COMPANY');
  $subject = "【{$co['name']}】卸パック 請求書（{$o['invoice_no']}）";
  $body = "{$o['name']} 様\n\n{$co['name']}です。卸パックのお申し込みありがとうございます。\n請求書を発行しましたので、下記よりご確認ください。\n\n"
    . "■ 請求書\n" . doc_url($o, 'invoice') . "\n（ブラウザの印刷から PDF 保存できます）\n\n"
    . "■ ご注文内容\n受注番号: {$o['id']}\nコース: {$c['label']} " . yen(order_course_price($o)) . ($o['photo'] ? "\n撮影オプション: " . yen(order_photo_fee($o)) : '') . "\n合計: " . yen($o['amount']) . "（税込）\n"
    . "お支払期限: " . jdate($o['due_date']) . "\n\n"
    . "■ お振込先\n" . bank_text() . "\n※振込手数料はお客様のご負担です。\n※振込名義は「{$o['name']}」でお願いします。\n\n"
    . "入金を確認しましたらメールでお知らせし、そこから" . cfg('SHIP_DAYS') . "日以内に発送します。\n"
    . ($o['photo'] ? "撮影オプション付きのため、発送前に撮影・採寸・画像加工をして画像データと一緒にお渡しします。\n" : '')
    . "\nご不明な点は" . cfg('SUPPORT_CHANNEL') . "またはこのメールへの返信でお知らせください。\n\n"
    . "--\n{$co['name']}\n{$co['address']}\nTEL {$co['tel']}\n{$co['mail']}\n特定商取引法に基づく表記: " . cfg('TOKUSHOHO_URL') . "\n";
  return [$subject, $body];
}

/** 請求書メールを送る（宛先・件名・本文を差し替え可）。送信履歴をログに残す */
function send_invoice_mail(array $o, ?string $to = null, ?string $subject = null, ?string $body = null, ?string $cc = null): bool {
  [$ds, $db] = invoice_mail_template($o);
  $to = $to ?: $o['email']; $subject = $subject ?: $ds; $body = $body ?: $db;
  $ok = send_mail($to, $subject, $body, null, $cc ?: null);
  update_order($o['id'], function (&$x) use ($to, $ok, $cc) { $x['log'][] = [date('Y-m-d H:i:s'), ($ok ? '請求書メール送信 → ' : '請求書メール送信失敗 → ') . $to . ($cc ? " (cc {$cc})" : '')]; });
  return $ok;
}

function mail_invoice(array $o): bool {
  $c = course_of($o);
  $ok = send_invoice_mail($o);
  send_mail(cfg('OWNER_MAIL'), "[卸パック受注] {$c['label']} {$o['name']} 様 ({$o['id']})",
    "新しい卸パックの申し込みがありました。\n\n受注番号: {$o['id']}\n氏名: {$o['name']}\n会社: {$o['company']}\nメール: {$o['email']}\nTEL: {$o['tel']}\n送付先: 〒{$o['zip']} {$o['address']}\nコース: {$c['label']}\n撮影オプション: " . ($o['photo'] ? 'あり（' . yen(order_photo_fee($o)) . '）' : 'なし') . "\n合計: " . yen($o['amount']) . "\n備考: {$o['note']}\n\n請求書: " . doc_url($o, 'invoice') . "\n管理画面: " . cfg('BASE_URL') . "/admin.php\n");
  return $ok;
}

function mark_paid(string $id): ?array {
  $o = update_order($id, function (&$o) {
    if ($o['status'] !== 'invoiced') return;
    $o['status'] = 'paid'; $o['paid_at'] = date('Y-m-d H:i:s');
    $o['ship_deadline'] = date('Y-m-d', strtotime('+' . (int)cfg('SHIP_DAYS') . ' days'));
    $o['log'][] = [date('Y-m-d H:i:s'), '入金確認'];
  });
  if ($o && $o['status'] === 'paid') {
    $co = cfg('COMPANY'); $c = course_of($o);
    send_mail($o['email'], "【{$co['name']}】ご入金を確認しました（{$o['id']}）",
      "{$o['name']} 様\n\n{$c['label']}のご入金を確認しました。ありがとうございます。\n"
      . jdate($o['ship_deadline']) . "までに発送し、発送時に納品書と追跡番号をメールでお送りします。\n\n受注番号: {$o['id']}\n送付先: 〒{$o['zip']} {$o['address']}\n送付先に変更がある場合は、このメールに返信でお知らせください。\n\n--\n{$co['name']}\n{$co['mail']}\n");
  }
  return $o;
}

/** 注文の明細行（請求書・納品書・メールで共通） */
function order_course_price(array $o): int { return (int)($o['course_price'] ?? course_of($o)['price']); }
function order_photo_fee(array $o): int { return (int)($o['photo_fee'] ?? ($o['photo'] ? course_of($o)['photo'] : 0)); }
function order_items(array $o): array {
  $c = course_of($o);
  $items = [['name' => $c['label'] . ' 一式', 'qty' => 1, 'unit' => order_course_price($o)]];
  if ($o['photo']) $items[] = ['name' => '撮影オプション（撮影・採寸・画像加工・画像データ納品）', 'qty' => 1, 'unit' => order_photo_fee($o)];
  return $items;
}

/** 納品書を1ボタンで生成。$items 省略時は注文どおり（コース＋撮影オプション） */
function make_delivery(string $id, ?array $items = null): ?array {
  return update_order($id, function (&$o) use ($items) {
    if ($items === null) $items = order_items($o);
    $total = 0; foreach ($items as $it) $total += (int)$it['qty'] * (int)$it['unit'];
    if (!$o['delivery_no']) $o['delivery_no'] = next_number('DN');
    $o['delivery_date'] = date('Y-m-d');
    $o['delivery_items'] = array_values($items);
    $o['delivery_total'] = $total;
    $o['log'][] = [date('Y-m-d H:i:s'), '納品書作成 ' . $o['delivery_no']];
  });
}

function mark_shipped(string $id, string $tracking): ?array {
  $o = update_order($id, function (&$o) use ($tracking) {
    $o['status'] = 'shipped'; $o['shipped_at'] = date('Y-m-d H:i:s'); $o['tracking'] = $tracking;
    $o['log'][] = [date('Y-m-d H:i:s'), '発送 ' . $tracking];
  });
  if ($o) {
    $co = cfg('COMPANY'); $c = course_of($o);
    $extra = ($o['delivery_total'] ?? 0) > $o['amount'] ? "\n※納品書の金額がご請求額を上回る分（" . yen($o['delivery_total'] - $o['amount']) . "）は別途ご案内します。\n" : '';
    send_mail($o['email'], "【{$co['name']}】発送しました（{$o['id']}）",
      "{$o['name']} 様\n\n{$c['label']}を発送しました。\n\n追跡番号: {$tracking}\n納品書: " . doc_url($o, 'delivery') . "\n{$extra}\n届いた商品に不備があった場合は、到着後7日以内にご連絡ください。\n\n--\n{$co['name']}\n{$co['mail']}\n");
  }
  return $o;
}

function status_label(string $s): string { return ['invoiced'=>'入金待ち','paid'=>'入金済・発送待ち','shipped'=>'発送済','cancelled'=>'キャンセル'][$s] ?? $s; }
