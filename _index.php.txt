// ライセンス案内メール（Webhook と管理画面の再送で共用）

  $from = env('LICENSE_MAIL_FROM'); $to = $to ?: ($lic['email'] ?? null);
  if (!$from || !$to) return false;
  $body = "CataScope をお申し込みいただきありがとうございます。

ライセンスキー: {$lic['key']}
プラン: {$plan}

【セットアップ】

3. 「パッケージ化されていない拡張機能を読み込む」で解凍したフォルダを選択
4. 次のリンクを開くと、ライセンスキーが自動で設定されます
   {$publicUrl}/activate?key={$lic['key']}
   （手動で設定する場合: ツールバーの CataScope アイコン → 設定 → 「ライセンスキー」に上記キーを貼り付け）

ご不明点は support@luxroute.net まで。
株式会社ストックス";
  return @mail($to, '【CataScope】ライセンスキーのご案内', $body, "From: $from\r\nReply-To: $from\r\nContent-Type: text/plain; charset=UTF-8", '-f ' . $from);
};

$auth = function (?string $kind) {
  $key = header_('x-license-key');
  $lic = $key ? findLicense($key) : null;
  if (!$lic) json(['error' => 'license_invalid'], 401);
  $usage = null;
  if ($kind) {
    $r = consume($key, $kind);
    if (!$r['ok']) json(['error' => $r['error'], 'usage' => $r['usage'] ?? null, 'plan' => $r['plan'] ?? null], 402);
    $usage = $r['usage'];
  }
  return [$lic, $usage];
};
$admin = function () { if (!env('ADMIN_TOKEN') || header_('x-admin-token') !== env('ADMIN_TOKEN')) json(['error' => 'forbidden'], 403); };

try {
  if ($path === '/' || $path === '/index.php') { header('Content-Type: text/plain'); echo 'CataScope server ok'; Db::close(); exit; }

  if ($path === '/api/license/me' && $method === 'GET') {
    [$lic] = $auth(null);
    $plan = effectivePlan($lic);
    $u = is_array($lic['usage']) ? ($lic['usage'][month()] ?? ['text' => 0, 'image' => 0]) : ['text' => 0, 'image' => 0];
    json(['plan' => $plan, 'usage' => $u, 'limits' => $plan ? PLANS[$plan] : null, 'trialEndsAt' => $lic['trialEndsAt']]);
  }

  if ($path === '/api/search' && $method === 'POST') {
    [$lic, $usage] = $auth('text');
    $b = body(); $q = trim((string)($b['q'] ?? '')); $num = max(1, min(50, (int)($b['num'] ?? 20)));
    if ($q === '') json(['error' => 'q required'], 400);
    try {
      // 方針(2026-10-01): Google 上のロットは言語違いの URL(nl/fr/sv…)で索引されていることが多いので言語を限定しない(site:catawiki.com のみ)。
      // Google の結果タイトルはカテゴリ名(例 "Bags for Sale in Online Auctions")で商品名を含まないため、タイトルでの関連判定はしない
      // (関連性は拡張がロットページを取得した後の実タイトル・型番で判定する)。品目語を付けて順位を上げる。
      // 利用者の検索回数は 1 回分のみ消費する。
      $brand = ['hermes' => 'Hermès', 'chloe' => 'Chloé', 'moet' => 'Moët', 'lancome' => 'Lancôme', 'cafe' => 'Café'];
      $words = preg_split('/\s+/', $q);
      $qb = implode(' ', array_map(fn($w) => $brand[mb_strtolower($w)] ?? $w, $words));
      $cat = detectCategory($qb);
      $qq = trim($qb . (($cat !== '' && !preg_match('/\\b' . $cat . '\\b/i', $qb)) ? ' ' . $cat : ''));
      // 短縮クエリ: 長いタイトル(色・付属品・形容語まで含む)は Google で 0 件になりやすい。
      // 第 1 群が不発なら「先頭 6 語 + 型番」に絞って同じ検索式で引き直す(SomaLi の結果件数に寄せるための保険)。
      $toks = preg_split('/\\s+/', $qb);
      $models = []; foreach ($toks as $t) if (preg_match('/^(?:[MNR]\\d{5}|[A-Z]{1,4}\\d{3,}[A-Z]{0,2}|[A-Z]{1,4}\\d*(?:[-\\/][A-Z0-9]+){1,3})$/', $t)) $models[] = $t;
      $drop = ['brown','black','white','red','blue','green','pink','beige','grey','gray','navy','yellow','orange','purple','ivory','khaki','bordeaux','camel','multicolor','hardware','handbag','shoulder','mini','boston','bosuton','kadena','ladies','mens','women','men','unisex','two-tone','yg','wg','pg','ss','combi'];
      $core = array_values(array_filter($toks, fn($t) => !in_array(mb_strtolower($t), $drop, true) && !in_array($t, $models, true)));
      $short = implode(' ', array_merge(array_slice($core, 0, 6), $models));
      $short = ($short !== '' && $short !== $qb) ? $short : '';
      $seen = []; $results = []; $calls = 0; $passes = [];
      // 2026-10 時点の SerpAPI/Google では site: 付き検索の 2 ページ目以降(start≥10)で site: が無視され
      // 無関係サイトが返る(total_results が 10 億超になる)。そのためページ送りは使わず、
      // 「1 ページ目だけ」を検索式のバリエーション(言語別サブパス等)で並列に引いて lotId で重複排除する。
      // filter=0 で「類似ページの省略」を無効化(Catawiki のロットページは互いに類似と判定され大量に省略される)。
      // inurl:/l/ は付けない(Google が "Missing: inurl" 扱いにして結果を 1 件に畳む)。num は Google が無視するので送らない。
      // engine: 既定 google_light(2026-10-01 実測: google は site: 検索が 20 秒タイムアウト/空応答/ site: 無視が頻発。
      // google_light は 8 秒前後で安定し件数も同等以上)。.env SERP_ENGINE で切替可。管理者は debug 用に body.engine で指定可。
      $engine = (effectivePlan($lic) === 'admin' && in_array($b['engine'] ?? '', ['google', 'google_light'], true)) ? $b['engine'] : (env('SERP_ENGINE') ?: 'google_light');
      $collect = function (array $queries) use (&$seen, &$results, &$calls, &$passes, $engine): void {
        $list = [];
        foreach ($queries as $qs) {
          $p = ['engine' => $engine, 'q' => $qs, 'hl' => 'en', 'gl' => 'us', 'filter' => '0'];
          $list[] = $p; $passes[] = ['q' => $qs, 'raw' => 0, 'added' => 0];
        }
        $t0 = microtime(true); $calls += count($list);
        try { $res = serpMulti($list, 20); } catch (Throwable $e) { $res = array_fill(0, count($list), $e); }
        $ms = (int)round((microtime(true) - $t0) * 1000);
        $base = count($passes) - count($list);
        foreach ($res as $i => $j) {
          $k = $base + $i; $passes[$k]['ms'] = $ms;
          if ($j instanceof Throwable) { $passes[$k]['err'] = $j->getMessage(); continue; }
          $passes[$k]['raw'] = count($j['organic_results'] ?? []);
          $passes[$k]['total'] = $j['search_information']['total_results'] ?? null;
          $passes[$k]['links'] = array_map(fn($r) => $r['link'] ?? '', array_slice($j['organic_results'] ?? [], 0, 3));
          foreach ($j['organic_results'] ?? [] as $r) {
            $link = $r['link'] ?? '';
            // 言語プレフィックス無し(catawiki.com/l/123…)の URL も拾う
            if (!preg_match('#^https?://(?:www\.)?catawiki\.com/(?:(?<lang>[a-z]{2}(?:-[a-z]{2})?)/)?l/(?<id>\d+)#', $link, $m)) continue;
            if (isset($seen[$m['id']])) continue;
            $seen[$m['id']] = true; $passes[$k]['added']++;
            // 言語を en に正規化(どの言語 URL でも同じロットが返る)
            $results[] = ['link' => 'https://www.catawiki.com/en/l/' . $m['id'], 'lotId' => $m['id'], 'title' => (string)($r['title'] ?? ''), 'snippet' => $r['snippet'] ?? '', 'thumbnail' => $r['thumbnail'] ?? '', 'lang' => (string)($m['lang'] ?? '')];
          }
        }
      };
      // 第 1 群(並列 4 本):
      //   ① SomaLi Search と同じ検索式 `<検索語> site:catawiki.com inurl:/l/ <品目語>`(同じ経路で同じ候補を得るため必ず含める)
      //   ② 全言語+品目語 / ③ 英語ロットページ限定 / ④ オランダ語(Catawiki 本国・索引が最も厚い)
      $somali = trim("$qb site:catawiki.com inurl:/l/" . ($cat !== '' ? " $cat" : ''));
      $collect([$somali, "site:catawiki.com $qq", "site:catawiki.com/en/l $qb", "site:catawiki.com/nl/l $qb"]);
      // 足りなければ第 2 群: 短縮クエリで SomaLi 式 + 英・蘭(短縮できない場合は 仏・独・伊 のロットページ)
      if (count($results) < max(8, (int)ceil($num / 2))) {
        if ($short !== '') {
          $somaliShort = trim("$short site:catawiki.com inurl:/l/" . ($cat !== '' ? " $cat" : ''));
          $collect([$somaliShort, "site:catawiki.com/en/l $short", "site:catawiki.com/nl/l $short", "site:catawiki.com/de/l $short"]);
        } else {
          $collect(["site:catawiki.com/fr/l $qb", "site:catawiki.com/de/l $qb", "site:catawiki.com/it/l $qb"]);
        }
      }
      // それでも足りなければ第 3 群: 短縮クエリの 仏・伊
      if ($short !== '' && count($results) < max(8, (int)ceil($num / 2))) {
        $collect(["site:catawiki.com/fr/l $short", "site:catawiki.com/it/l $short"]);
      }
      $out = ['results' => array_slice($results, 0, 50), 'query' => $q, 'searchQuery' => $qq, 'category' => $cat, 'shortQuery' => $short, 'calls' => $calls, 'engine' => $engine, 'usage' => $usage];
      if (!empty($b['debug']) && effectivePlan($lic) === 'admin') $out['passes'] = $passes;
      json($out);
    } catch (Throwable $e) { refund($lic['key'], 'text'); json(['error' => $e->getMessage()], 502); }
  }

  // ---- 落札データベース（外部検索を使わないので検索回数は消費しない） ----
  if ($path === '/api/lots/search' && $method === 'POST') {
    $auth(null); $b = body(); $q = trim((string)($b['q'] ?? '')); $num = max(1, min(100, (int)($b['num'] ?? 40)));
    if ($q === '') json(['error' => 'q required'], 400);
    try { $lots = Lots::search($q, $num, !empty($b['closedOnly'])); $st = Lots::stats(); }
    catch (Throwable $e) { json(['error' => 'db_error', 'detail' => $e->getMessage()], 500); }
    json(['lots' => $lots, 'dbTotal' => (int)($st['total'] ?? 0)]);
  }
  if ($path === '/api/lots/ingest' && $method === 'POST') {
    $auth(null); $b = body(); $lots = is_array($b['lots'] ?? null) ? array_slice($b['lots'], 0, 60) : [];
    try { $n = Lots::ingest($lots, (string)($b['source'] ?? 'ext')); } catch (Throwable $e) { json(['error' => 'db_error', 'detail' => $e->getMessage()], 500); }
    json(['ok' => true, 'stored' => $n]);
  }
  if ($path === '/admin/lots/stats' && $method === 'GET') { $admin(); json(Lots::stats()); }

  // ---- 巡回クローラー（拡張がブラウザの IP で取得し、サーバーは配布と回収だけ行う） ----
  if (strpos($path, '/api/crawl/') === 0 && !class_exists('Crawl')) json(['error' => 'crawler_not_deployed'], 503);
  if ($path === '/api/crawl/next' && $method === 'POST') {
    [$lic] = $auth(null); $b = body();
    try { $tasks = Crawl::next(substr($lic['key'], 0, 11), (int)($b['n'] ?? 2)); } catch (Throwable $e) { json(['error' => 'db_error', 'detail' => $e->getMessage()], 500); }
    json(['tasks' => $tasks]);
  }
  if ($path === '/api/crawl/done' && $method === 'POST') {
    [$lic] = $auth(null); $b = body();
    try { $r = Crawl::done((int)($b['taskId'] ?? 0), (string)($b['kind'] ?? ''), is_array($b['payload'] ?? null) ? $b['payload'] : [], isset($b['error']) ? (string)$b['error'] : null); }
    catch (Throwable $e) { json(['error' => 'db_error', 'detail' => $e->getMessage()], 500); }
    json(['ok' => true] + $r);
  }
  if ($path === '/admin/crawl/seed' && $method === 'POST') { $admin(); json(['seeded' => Crawl::seed(), 'stats' => Crawl::stats()]); }
  if ($path === '/admin/crawl/stats' && $method === 'GET') { $admin(); json(Crawl::stats()); }

  if ($path === '/api/lens' && $method === 'POST') {
    [$lic, $usage] = $auth('image');
    $b = body(); $img = trim((string)($b['imageUrl'] ?? ''));
    if ($img === '') json(['error' => 'imageUrl required'], 400);
    try {
      $j = serp(['engine' => 'google_lens', 'url' => $img, 'hl' => 'en']);
      $vm = $j['visual_matches'] ?? [];
      $results = array_map(fn($v) => ['link' => $v['link'] ?? '', 'title' => $v['title'] ?? '', 'thumbnail' => $v['thumbnail'] ?? '', 'source' => $v['source'] ?? ''], $vm);
      $titles = array_values(array_filter(array_map(fn($v) => $v['title'] ?? '', array_slice($vm, 0, 5))));
      json(['results' => $results, 'titles' => $titles, 'usage' => $usage]);
    } catch (Throwable $e) { refund($lic['key'], 'image'); json(['error' => $e->getMessage()], 502); }
  }

  if ($path === '/api/upload' && $method === 'POST') {
    $auth(null);
    $b = body();
    if (!preg_match('#^data:image/(png|jpeg);base64,(.+)$#s', (string)($b['dataUrl'] ?? ''), $m)) json(['error' => 'invalid dataUrl'], 400);
    $bin = base64_decode($m[2], true);
    if ($bin === false || strlen($bin) > 4 * 1024 * 1024) json(['error' => 'too large'], 413);
    // 24 時間より古い切り抜きを掃除
    foreach (glob(DATA_DIR . '/uploads/*') ?: [] as $f) if (filemtime($f) < time() - 86400) @unlink($f);
    $name = bin2hex(random_bytes(12)) . ($m[1] === 'png' ? '.png' : '.jpg');
    file_put_contents(DATA_DIR . '/uploads/' . $name, $bin);
    json(['url' => "$publicUrl/u/$name"]);
  }

  if (preg_match('#^/u/([a-f0-9]{24}\.(png|jpg))$#', $path, $m)) {
    $f = DATA_DIR . '/uploads/' . $m[1];
    if (!is_file($f)) { http_response_code(404); exit; }
    header('Content-Type: ' . ($m[2] === 'png' ? 'image/png' : 'image/jpeg')); header('Cache-Control: public, max-age=3600');
    readfile($f); Db::close(); exit;
  }

  if ($path === '/admin/licenses' && $method === 'POST') { $admin(); $b = body(); $plan = $b['plan'] ?? 'trial'; if (!isset(PLANS[$plan])) json(['error' => 'unknown_plan', 'plans' => array_keys(PLANS)], 400); json(createLicense($b['email'] ?? null, $plan)); }
  if ($path === '/admin/plans' && $method === 'GET') { $admin(); json(PLANS); }
  if ($path === '/admin/licenses' && $method === 'GET') { $admin(); json(array_values(Db::licenses())); }
  // 設定の有無だけを返す（値は返さない）
  if ($path === '/admin/licenses/mail' && $method === 'POST') {
    $admin(); $b = body(); $lic = Db::licenses()[(string)($b['key'] ?? '')] ?? null;
    if (!$lic) json(['error' => 'not_found'], 404);
    $ok = $sendLicenseMail($lic, (string)$lic['plan'], (string)($b['email'] ?? '') ?: null);
    json(['ok' => $ok]);
  }
  if ($path === '/admin/env-status' && $method === 'GET') { $admin(); $st = []; foreach (['PUBLIC_URL','SERPAPI_KEY','ADMIN_TOKEN','STRIPE_SECRET_KEY','STRIPE_WEBHOOK_SECRET','STRIPE_PRICE_STD_M','STRIPE_PRICE_STD_Y','STRIPE_PRICE_PRO_M','STRIPE_PRICE_PRO_Y','LICENSE_MAIL_FROM'] as $k) { $v = (string)env($k, ''); $st[$k] = ['set' => $v !== '', 'len' => strlen($v), 'prefix' => $v !== '' ? substr($v, 0, 3) : '']; } $st['envKeys'] = array_values(array_filter(array_keys($_ENV), fn($k) => !in_array($k, ['argv','argc']))); json($st); }
  if (preg_match('#^/admin/licenses/([A-Z0-9-]+)$#', $path, $m) && $method === 'POST') { $admin(); $r = updateLicense($m[1], body()); $r ? json($r) : json(['error' => 'not_found'], 404); }

  if ($path === '/webhook/stripe' && $method === 'POST') {
    $secret = env('STRIPE_SECRET_KEY'); $whsec = env('STRIPE_WEBHOOK_SECRET');
    if (!$secret || !$whsec) json(['error' => 'stripe not configured'], 501);
    $raw = file_get_contents('php://input') ?: '';
    // 署名検証（Stripe-Signature: t=...,v1=...）
    $parts = []; foreach (explode(',', header_('stripe-signature')) as $p) { [$k, $v] = array_pad(explode('=', $p, 2), 2, ''); $parts[$k][] = $v; }
    $t = $parts['t'][0] ?? ''; $ok = false;
    foreach ($parts['v1'] ?? [] as $sig) if (hash_equals(hash_hmac('sha256', "$t.$raw", $whsec), $sig)) $ok = true;
    if (!$ok || abs(time() - (int)$t) > 300) { http_response_code(400); echo 'bad signature'; Db::close(); exit; }
    $ev = json_decode($raw, true) ?: [];
    $priceToPlan = array_filter([env('STRIPE_PRICE_STD_M', '') => 'standard', env('STRIPE_PRICE_STD_Y', '') => 'standard_annual', env('STRIPE_PRICE_PRO_M', '') => 'pro', env('STRIPE_PRICE_PRO_Y', '') => 'pro_annual'], fn($v, $k) => $k !== '', ARRAY_FILTER_USE_BOTH);
    $stripeGet = function (string $p) use ($secret) { $ch = curl_init("https://api.stripe.com/v1/$p"); curl_setopt_array($ch, [CURLOPT_RETURNTRANSFER => true, CURLOPT_USERPWD => $secret . ':', CURLOPT_TIMEOUT => 20]); $r = curl_exec($ch); curl_close($ch); return json_decode((string)$r, true) ?: []; };
    $type = $ev['type'] ?? ''; $obj = $ev['data']['object'] ?? [];
    if ($type === 'checkout.session.completed') {
      $sub = $stripeGet('subscriptions/' . ($obj['subscription'] ?? ''));
      $plan = $priceToPlan[$sub['items']['data'][0]['price']['id'] ?? ''] ?? 'standard';
      $lic = createLicense($obj['customer_details']['email'] ?? null, $plan, $obj['customer'] ?? null, $obj['subscription'] ?? null);
      error_log('[license issued] ' . $lic['key'] . ' ' . ($lic['email'] ?? '') . ' ' . $plan);
      $sendLicenseMail($lic, $plan);
    }
    if ($type === 'customer.subscription.deleted' || ($type === 'customer.subscription.updated' && !in_array($obj['status'] ?? '', ['active', 'trialing'], true))) {
      $lic = findLicenseByCustomer($obj['customer'] ?? ''); if ($lic) updateLicense($lic['key'], ['active' => false]);
    }
    if ($type === 'customer.subscription.updated' && in_array($obj['status'] ?? '', ['active', 'trialing'], true)) {
      $lic = findLicenseByCustomer($obj['customer'] ?? ''); $plan = $priceToPlan[$obj['items']['data'][0]['price']['id'] ?? ''] ?? null;
      if ($lic) updateLicense($lic['key'], ['active' => true] + ($plan ? ['plan' => $plan] : []));
    }
    json(['received' => true]);
  }

  json(['error' => 'not_found'], 404);
} catch (Throwable $e) {
  json(['error' => $e->getMessage()], 500);
}
